\pmheader{LITERATURE REVIEW}{13 / 22}
\par\pmtitle{From forecast performance to market mechanisms}
\begin{pmbody}
{\color{pmGrey}\fontsize{20}{27}\selectfont Research increasingly asks which traders and institutions make prices informative.\par}
\begin{itemize}\pmspreaditems
  \item Early research establishes the foundations and conditions for information aggregation.
  \item Recent empirical work connects trader composition with profitability, liquidity provision, and forecast accuracy.
  \item A central unresolved question is whether changes in participation cause changes in market quality.
\end{itemize}
\begin{pmcallout}[]
  This review identifies potential contribution areas without committing to a particular research direction.
\end{pmcallout}
\end{pmbody}
\end{frame}
