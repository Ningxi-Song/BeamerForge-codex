      (0.2pt,4.6pt) .. controls (0.8pt,4.95pt) and (1.25pt,4.15pt) .. (1.55pt,3.55pt)

% Draw an irregular pebble at a given coordinate.  The shape is a
% slightly distorted ellipse with a deterministic, index-driven seed.
\newcommand{\pebbleat}[3]{%
  % Five radial points with random radii and angles around the center.
  \pgfmathsetmacro{\rA}{#2*(0.72+rnd*0.56)}%
  \pgfmathsetmacro{\rB}{#2*(0.72+rnd*0.56)}%
  \pgfmathsetmacro{\rC}{#2*(0.72+rnd*0.56)}%
  \pgfmathsetmacro{\rD}{#2*(0.72+rnd*0.56)}%
  \pgfmathsetmacro{\rE}{#2*(0.72+rnd*0.56)}%
  \pgfmathsetmacro{\aA}{rnd*50}%
  \pgfmathsetmacro{\aB}{\aA+65+rnd*35}%
  \pgfmathsetmacro{\aC}{\aB+65+rnd*35}%
  \pgfmathsetmacro{\aD}{\aC+65+rnd*35}%
  \pgfmathsetmacro{\aE}{\aD+65+rnd*35}%
  \fill[#3] plot[smooth cycle,tension=0.9] coordinates {
    ($(#1)+(\aA:\rA pt)$)
    ($(#1)+(\aB:\rB pt)$)
    ($(#1)+(\aC:\rC pt)$)
    ($(#1)+(\aD:\rD pt)$)
    ($(#1)+(\aE:\rE pt)$)
  };%
}

\newcommand{\progressdots}{%
  \begin{tikzpicture}[remember picture,overlay]
    \ifnum\inserttotalframenumber>0
      \pgfmathsetmacro{\progratio}{\insertframenumber/\inserttotalframenumber}%
    \else
      \def\progratio{0}%
    \fi
    % Geometry (dots hug the top edge; title is pushed down to create a gap)
    \def\dotY{-0.20cm}
    \def\dotL{0.06\paperwidth}
    \def\dotR{0.94\paperwidth}
    \pgfmathsetmacro{\totaldots}{30}
    \pgfmathsetmacro{\currentdot}{max(1,round(\progratio*\totaldots))}
    % Draw the pebble row
    \foreach \i in {1,...,\totaldots} {
      \pgfmathsetmacro{\dotx}{\dotL + (\i-1)*(\dotR-\dotL)/(\totaldots-1)}
      \coordinate (D) at ([xshift=\dotx,yshift=\dotY]current page.north west);
      \pgfmathsetseed{\i*137}
      \ifnum\i=\currentdot
        % current position: larger pebble
        \pebbleat{D}{2.6}{bfblack}
      \else
        \ifnum\i<\currentdot
          % completed: small black pebble
          \pebbleat{D}{1.25}{bfblack}
        \else
          % upcoming: small gray pebble
          \pebbleat{D}{1.25}{bfgray}
        \fi
      \fi
    }
  \end{tikzpicture}%
}

\section{Introduction}

\begin{frame}{Motivation}
  This template is designed for academic talks that demand:
  \begin{itemize}
    \item A strict black-and-white palette with no color noise.
    \item Classical EB Garamond headings over Latin Modern body and math.
    \item Arrow bullets and a clean geometric motif.
    \item Generous whitespace and a clear visual hierarchy.
  \end{itemize}
  \vspace{0.5cm}
  The title page keeps a clean centered layout, free of decorative accents.
\end{frame}

\begin{frame}{Core design decisions}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Typography}
      \begin{itemize}
        \item EB Garamond (classical serif) for headings.
        \item Latin Modern Roman for body text.
        \item Latin Modern Math for equations.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Color}
      \begin{itemize}
        \item Background: pure white.
        \item Structure: near-black.
        \item Accents: light gray rules and block fills.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.6cm}
  All bullets use a sharp arrow marker for a forceful look.
\end{frame}

\section{Mathematical content}

\begin{frame}{Equations}
  The Latin Modern math font keeps formulas visually consistent with the text:
  \[
    \min_{x \in \mathbb{R}^n} \; f(x)
    \quad \text{subject to} \quad
    g_i(x) \leq 0, \; i = 1, \dots, m.
  \]
  \vspace{0.3cm}
  For a linear model, the ordinary least-squares estimator is
  \[
    \hat{\beta} = (X^{\top} X)^{-1} X^{\top} y.
  \]
\end{frame}

\begin{frame}{Theorem and proof sketch}
  \begin{block}{Theorem (Consistency)}
    Under standard regularity conditions, the estimator \(\hat{\beta}_n\) converges in probability to the true parameter \(\beta_0\).
  \end{block}
  \vspace{0.3cm}
  \begin{alertblock}{Key insight}
    Compactness of the parameter space and uniform convergence of the objective function are sufficient.
  \end{alertblock}
  \vspace{0.3cm}
  \begin{exampleblock}{Example}
    If \(\varepsilon_i \sim \mathcal{N}(0, \sigma^2)\), inference proceeds via the usual \(t\)- and \(F\)-statistics.
  \end{exampleblock}
\end{frame}
